import java.util.*;
public class VanyaAndCubes{
    public static void main(String[]args){
        Scanner sc=new Scanner(System.in);
        int a=sc.nextInt();
        int count=0;
        int i=1;
        int sum=1;
        while(a>0){
            a-=sum;
            i++;
            sum+=i;
            count++;
            if(sum>a){
                break;
            }
        }
        System.out.println(count);
        sc.close();
    }
}
