import java.util.Scanner;

public class LoveStory {
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        String s="codeforces";
        int t=sc.nextInt();
        while(t-->0){
            String p=sc.next();
            int i=0;
            int count=0;
            while(i<s.length()){
                if(s.charAt(i)!=p.charAt(i)){
                    count++;
                }
                i++;
            }
            System.out.println(count);
        }
        sc.close();
    }
}
