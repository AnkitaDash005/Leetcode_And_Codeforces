import java.util.*;
public class AmbitiousKid {
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt();
        int []arr=new int[n];
        arr[0]=Math.abs(sc.nextInt());
        int min=arr[0];
        int i=1;
        while(i<n){
            arr[i]=Math.abs(sc.nextInt());
            min=Math.min(min,arr[i]);
            i++;
        }
        System.out.println(min);
        sc.close();
    }
}
