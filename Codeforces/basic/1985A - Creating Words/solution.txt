import java.util.Scanner;

public class CreatingWords {
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        int t=sc.nextInt();
        while(t-->0){
            String a=sc.next();
            String b=sc.next();
            String temp1=""+a.charAt(0);
            String temp2=""+b.charAt(0);
            int i=1;
            while(i<a.length()){
                temp1+=b.charAt(i);
                temp2+=a.charAt(i);
                i++;
            }
            System.out.println(temp2+" "+temp1);


            
        }
        sc.close();
    }
}
