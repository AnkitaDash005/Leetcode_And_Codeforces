import java.util.Scanner;

public class Calculating_function {
    public static void main(String[]args){
        Scanner sc=new Scanner(System.in);
        long k=sc.nextLong();
        if (k % 2 == 0) {
            System.out.println(k / 2);
        } else {
            
            System.out.println(-(k + 1) / 2);
        }
        sc.close();
    }
}
